    Chapter 07

    抽樣

    用 Python 練習簡單隨機抽樣、分層抽樣、集群抽樣，以及中央極限定理的模擬。

    本章學習目標

    • 會用 Python 從資料中抽出樣本。
    • 會實作簡單隨機抽樣、分層抽樣、集群抽樣與便利抽樣示意。
    • 會用重複抽樣模擬樣本平均數的抽樣分配。
    • 會用圖形觀察中央極限定理：樣本平均數會逐漸接近常態分配。

    0. 套件與範例資料

    這一章直接用一份模擬的學生資料做練習。 每一列代表一位學生，包含學院、年級、班級、每週讀書時數與是否回覆問卷。 

    import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

np.random.seed(42)

N = 20000

population = pd.DataFrame({
    "student_id": np.arange(1, N + 1),
    "college": np.random.choice(
        ["管理學院", "工程學院", "電機學院", "人社學院"],
        size=N,
        p=[0.25, 0.35, 0.25, 0.15]
    ),
    "grade": np.random.choice([1, 2, 3, 4], size=N),
    "class_id": np.random.choice([f"C{i:03d}" for i in range(1, 301)], size=N),
})

college_effect = {
    "管理學院": 1.0,
    "工程學院": 2.0,
    "電機學院": 2.5,
    "人社學院": 0.5
}

population["study_hours"] = (
    np.random.normal(loc=14, scale=5, size=N)
    + population["college"].map(college_effect)
)

population["study_hours"] = population["study_hours"].clip(lower=0)

population["responded"] = np.random.choice(
    [1, 0],
    size=N,
    p=[0.35, 0.65]
)

population.head()

    population.shape

true_mean = population["study_hours"].mean()
true_std = population["study_hours"].std(ddof=0)

print("母體平均數：", round(true_mean, 3))
print("母體標準差：", round(true_std, 3))

    1. 簡單隨機抽樣

    簡單隨機抽樣就是從母體中隨機抽出指定數量的觀察值。 在 Pandas 中可以直接使用 .sample()。 

    sample_srs = population.sample(n=100, random_state=42)

sample_srs.head()

    計算樣本平均數：

    srs_mean = sample_srs["study_hours"].mean()

print("簡單隨機樣本平均數：", round(srs_mean, 3))
print("母體平均數：", round(true_mean, 3))
print("差距：", round(srs_mean - true_mean, 3))

    有放回與無放回抽樣

    replace=False 表示抽過的人不會再被抽到； replace=True 表示抽完後放回，可能重複抽到同一筆資料。 

    sample_without_replacement = population.sample(
    n=10,
    replace=False,
    random_state=1
)

sample_with_replacement = population.sample(
    n=10,
    replace=True,
    random_state=1
)

sample_with_replacement["student_id"].value_counts()

selected_ids = student_ids.sample(n=100, random_state=7)

    3. 分層抽樣

    分層抽樣會先依某個欄位分組，例如學院或年級， 再從每一層中抽出樣本。這裡示範按「學院」分層。

    每層抽一樣多人

    sample_strat_equal = (
    population
    .groupby("college", group_keys=False)
    .apply(lambda x: x.sample(n=30, random_state=42))
)

sample_strat_equal["college"].value_counts()

    依照各層比例抽樣

    n_total = 200

college_prop = population["college"].value_counts(normalize=True)

sample_sizes = (college_prop * n_total).round().astype(int)

sample_sizes
    sample_strat_prop = (
    population
    .groupby("college", group_keys=False)
    .apply(lambda x: x.sample(
        n=sample_sizes.loc[x.name],
        random_state=42

)

sample_strat_prop["college"].value_counts(normalize=True)

    比較母體與比例分層樣本的學院比例：

    "樣本比例": sample_strat_prop["college"].value_counts(normalize=True)

    5. 便利抽樣與自願回覆偏差示意

    便利抽樣不是嚴格隨機抽樣。這裡用「只看有回覆問卷的人」示範， 即使樣本數很大，也可能和母體有差距。 

    convenience_pool = population[population["responded"] == 1]

sample_convenience = convenience_pool.sample(
    n=1000,
    random_state=42
)

print("母體平均讀書時數：", round(population["study_hours"].mean(), 3))
print("便利樣本平均讀書時數：", round(sample_convenience["study_hours"].mean(), 3))
print("回覆者比例：", round(convenience_pool.shape[0] / population.shape[0], 3))
    樣本數大不代表一定是好樣本。若抽樣方式本身有偏差， 增加樣本數只能讓估計更穩定地偏掉，聽起來很殘酷，但統計就是這麼不講武德。

    6. 重複抽樣：建立樣本平均數的抽樣分配

    抽一次只會得到一個樣本平均數。 若重複抽很多次，把每次的樣本平均數收集起來，就能看到樣本平均數的抽樣分配。 

    def simulate_sample_means(data, sample_size, n_sim=5000, random_state=42):
    rng = np.random.default_rng(random_state)
    values = data.to_numpy()
    means = []

    for _ in range(n_sim):
        sample = rng.choice(values, size=sample_size, replace=False)
        means.append(sample.mean())

    return np.array(means)

    以樣本數 n = 25 重複抽樣 5,000 次：

    sample_means_25 = simulate_sample_means(
    population["study_hours"],
    sample_size=25,
    n_sim=5000,
    random_state=42
)

sample_means_25[:10]

    畫出樣本平均數分布：

plt.hist(sample_means_25, bins=40)
plt.axvline(true_mean, linestyle="--", label="Population Mean")
plt.title("Sampling Distribution of Sample Mean, n = 25")

    7. 樣本數變大，標準誤變小

    這裡用不同樣本數重複抽樣，比較樣本平均數的分布寬度。 

    sample_sizes = [4, 25, 100, 400]
results = {}

for n in sample_sizes:
    results[n] = simulate_sample_means(
        population["study_hours"],
        sample_size=n,
        n_sim=5000,
        random_state=42 + n
    )

summary = pd.DataFrame({
    "sample_size": sample_sizes,
    "simulated_mean": [results[n].mean() for n in sample_sizes],
    "simulated_se": [results[n].std(ddof=1) for n in sample_sizes],
    "theoretical_se": [true_std / np.sqrt(n) for n in sample_sizes]
})

summary

    把不同樣本數的抽樣分配畫在一起：

    plt.figure(figsize=(9, 5))

for n in sample_sizes:
    plt.hist(
        results[n],
        bins=40,
        density=True,
        label=f"n = {n}"
    )

plt.axvline(true_mean, linestyle="--", label="Population Mean")
plt.title("Sample Size and Sampling Distribution")
plt.xlabel("Sample Mean")
plt.ylabel("Density")
plt.legend()
plt.show()
    可以觀察到：樣本數愈大，樣本平均數愈集中在母體平均數附近。 也就是說，樣本平均數會比較不晃。

    8. 中央極限定理模擬

    這個範例故意使用非常右偏的指數分配當母體。 原始資料不是常態，但樣本平均數的分布會隨著樣本數變大而越來越像常態。

    建立右偏母體

    np.random.seed(123)

skewed_population = np.random.exponential(
    scale=10,
    size=100000
)

plt.figure(figsize=(8, 5))
plt.hist(skewed_population, bins=80)
plt.title("Original Population: Exponential Distribution")
plt.xlabel("Value")
plt.ylabel("Frequency")
plt.show()

    模擬不同 n 的樣本平均數

    def clt_simulation(population_values, sample_size, n_sim=5000, random_state=0):
    rng = np.random.default_rng(random_state)
    means = []

    for _ in range(n_sim):
        sample = rng.choice(population_values, size=sample_size, replace=True)
        means.append(sample.mean())

    return np.array(means)

n_list = [1, 2, 5, 30, 100]

clt_results = {
    n: clt_simulation(
        skewed_population,
        sample_size=n,
        n_sim=5000,
        random_state=100 + n
    )
    for n in n_list
}

    分別畫出不同樣本數下的樣本平均數分布：

    for n in n_list:
    plt.figure(figsize=(8, 5))
    plt.hist(clt_results[n], bins=50, density=True)
    plt.axvline(skewed_population.mean(), linestyle="--", label="Population Mean")
    plt.title(f"CLT Simulation: Distribution of Sample Means, n = {n}")
    plt.xlabel("Sample Mean")
    plt.ylabel("Density")
    plt.legend()
    plt.show()

    用表格比較平均數與標準誤：

    clt_summary = pd.DataFrame({
    "sample_size": n_list,
    "mean_of_sample_means": [clt_results[n].mean() for n in n_list],
    "sd_of_sample_means": [clt_results[n].std(ddof=1) for n in n_list],
    "theoretical_se": [skewed_population.std(ddof=0) / np.sqrt(n) for n in n_list]
})

    9. 樣本平均數機率計算範例

    假設某飲品消費金額的母體平均為 4.18，母體標準差為 0.84。 現在抽 50 位顧客，想計算樣本平均消費金額大於等於 4.26 的機率。 

    from scipy.stats import norm

mu = 4.18
sigma = 0.84
n = 50
x_bar = 4.26

se = sigma / np.sqrt(n)
z = (x_bar - mu) / se

prob = 1 - norm.cdf(z)

print("標準誤 SE =", round(se, 4))
print("z 值 =", round(z, 4))
print("P(X_bar >= 4.26) =", round(prob, 4))

    也可以直接寫成一個函式：

    def prob_sample_mean_greater_equal(x_bar, mu, sigma, n):
    se = sigma / np.sqrt(n)
    z = (x_bar - mu) / se
    return 1 - norm.cdf(z)

prob_sample_mean_greater_equal(
    x_bar=4.26,
    mu=4.18,
    sigma=0.84,
    n=50
)

    練習題

    1. 從 population 中簡單隨機抽出 300 筆資料，計算 study_hours 的平均數。
    2. 依照 grade 分層，每個年級各抽 50 位學生。
    3. 隨機抽出 5 個 class_id，取出這些班級的所有學生。
    4. 將樣本數改成 n = 10, 50, 200，重跑中央極限定理模擬並比較圖形。
    5. 使用 scipy.stats.norm 計算：若 mu = 70、sigma = 15、n = 100，樣本平均數大於 73 的機率。

    本章重點整理

    • DataFrame.sample() 可以快速完成簡單隨機抽樣。
    • groupby().apply() 可以實作分層抽樣。
    • 集群抽樣可以先抽群體 ID，再篩選出群體內所有資料。
    • 便利抽樣容易有偏差，即使樣本數很大也不代表有代表性。
    • 重複抽樣可以模擬樣本平均數的抽樣分配。
    • 樣本數越大，樣本平均數的標準誤越小。
    • 中央極限定理可以用模擬觀察：即使母體不是常態，樣本平均數的分布也會逐漸接近常態。
